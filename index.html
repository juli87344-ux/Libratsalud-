import webpush from "npm:web-push@3.6.7"; import { createClient } from "npm:@supabase/supabase-js@2"; export default { fetch: async (req: Request) => { try { const publicKey = Deno.env.get("VAPID_PUBLIC_KEY"); const privateKey = Deno.env.get("VAPID_PRIVATE_KEY"); const supabaseUrl = Deno.env.get("SUPABASE_URL"); const serviceKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY"); if (!publicKey || !privateKey || !supabaseUrl || !serviceKey) { throw new Error("Faltan variables necesarias"); } let payload = { title: "Libratsalud", body: "Nueva actualización en Libratsalud" }; try { const recibido = await req.json(); if (recibido?.title) { payload.title = recibido.title; } if (recibido?.body) { payload.body = recibido.body; } } catch { // Si no llega un cuerpo JSON, usa el mensaje predeterminado. } webpush.setVapidDetails( "mailto:admin@libratsalud.app", publicKey, privateKey ); const supabase = createClient(supabaseUrl, serviceKey); const { data, error } = await supabase .from("notificaciones") .select("id, endpoint, p256dh, auth") .eq("activo", true); if (error) throw error; const resultados = []; for (const s of data ?? []) { try { await webpush.sendNotification( { endpoint: s.endpoint, keys: { p256dh: s.p256dh, auth: s.auth } }, JSON.stringify(payload) ); resultados.push({ id: s.id, enviado: true }); } catch (e: any) { if (e?.statusCode === 410 || e?.statusCode === 404) { await supabase .from("notificaciones") .update({ activo: false }) .eq("id", s.id); } resultados.push({ id: s.id, enviado: false, error: String(e), statusCode: e?.statusCode ?? null, body: e?.body ?? null }); } } return Response.json({ ok: true, resultados }); } catch (e) { return Response.json( { ok: false, error: String(e) }, { status: 500 } ); } } };
